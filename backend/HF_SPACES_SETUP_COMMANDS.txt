# 1.1) Optional but recommended: log in via HF CLI once (stores token securely)

# Paste a token with write permission when prompted.

# 2) Initialize Git LFS (already done on your machine)
git lfs install

# 3) Clone your HF Space repo
git clone https://huggingface.co/spaces/abk4tech/orthoai-backend hf-space-backend

# If clone prompts for credentials:
# Username: your HF username (abk4tech)
# Password: your HF access token (write permission)

# 4) Sync backend contents into the Space repo root (excluding local/runtime folders)
robocopy backend hf-space-backend /E /XD venv uploads __pycache__ .pytest_cache .mypy_cache .vscode .idea /XF backend_structure.txt

# 5) Enter the Space repo
cd hf-space-backend

# 6) Track model files with LFS
git lfs track "*.onnx"
git lfs track "*.pt"

# 7) Commit and push to Hugging Face
git add .gitattributes .
git commit -m "Deploy OrthoAI backend to Hugging Face Space"
git push origin main

# If push asks for password, use the HF access token again.
